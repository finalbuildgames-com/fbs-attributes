cmake_minimum_required(VERSION 3.16)
project(FinalBuildAttributes VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_attributes src/attributes.c)
add_library(fbs::attributes ALIAS fbs_attributes)
set_target_properties(fbs_attributes PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME attributes)
target_include_directories(fbs_attributes PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_attributes PUBLIC m)
  target_compile_options(fbs_attributes PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_attributes tests/test_attributes.c)
  set_target_properties(fbs_test_attributes PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_attributes PRIVATE fbs::attributes)
  if(NOT MSVC)
    target_compile_options(fbs_test_attributes PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME attributes COMMAND fbs_test_attributes)
  set_tests_properties(attributes PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_attributes_example examples/basic.c)
  set_target_properties(fbs_attributes_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_attributes_example PRIVATE fbs::attributes)
  if(BUILD_TESTING)
    add_test(NAME attributes_example COMMAND fbs_attributes_example)
  endif()
endif()
install(TARGETS fbs_attributes EXPORT FinalBuildAttributesTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildAttributesTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildAttributes)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-attributes)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-attributes/upstream OPTIONAL)
